"""Find a board inside a bigger picture (for example a photo of a screen).
Used only as a fallback when the normal screenshot finder fails."""
import os, sys, itertools
import cv2
import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.join(HERE, "..", "scripts"))
import boardlib as B
from ai_recognize import recognize_ai


def _candidates(img, K=8):
    """Square-ish blobs made of two of the picture's main colours."""
    H, W = img.shape[:2]
    sc = 800.0 / max(H, W)
    small = cv2.resize(img, (int(W * sc), int(H * sc)), interpolation=cv2.INTER_AREA)
    lab = cv2.cvtColor(small, cv2.COLOR_BGR2LAB)
    h, w = lab.shape[:2]
    data = lab.reshape(-1, 3).astype("float32")
    crit = (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 15, 1.0)
    cv2.setRNGSeed(1)
    _, _, centers = cv2.kmeans(data[::4], K, None, crit, 1, cv2.KMEANS_PP_CENTERS)
    labels = ((data[:, None, :] - centers[None, :, :]) ** 2).sum(2).argmin(1).reshape(h, w)
    seen, out = set(), []
    for i, j in itertools.combinations(range(K), 2):
        m = (((labels == i) | (labels == j)) * 255).astype("uint8")
        m = cv2.morphologyEx(m, cv2.MORPH_CLOSE, np.ones((9, 9), np.uint8))
        cnts, _ = cv2.findContours(m, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        for c in cnts:
            x, y, bw, bh = cv2.boundingRect(c)
            if bw < 0.2 * w or not (0.9 < bw / bh < 1.1):
                continue
            if cv2.contourArea(c) / float(bw * bh) < 0.7:
                continue
            key = (x // 12, y // 12, bw // 12)
            if key not in seen:
                seen.add(key)
                out.append((x / sc, y / sc, bw / sc))
    return out


def _quality(img, box):
    """How sure is the AI about this crop? (higher = better)"""
    x, y, s = box
    H, W = img.shape[:2]
    if x < 0 or y < 0 or s < 100 or x + s > W or y + s > H:
        return -1.0
    _, conf = recognize_ai(img[y:y+s, x:x+s])
    flat = np.array(conf).ravel()
    return 0.5 * float(flat.min()) + 0.5 * float(flat.mean())


def _polish(img, box):
    """Nudge x, y, size a few pixels at a time while the AI gets more confident."""
    best = tuple(int(round(v)) for v in box)
    bq = _quality(img, best)
    for step in (8, 4, 2):
        improved = True
        while improved:
            improved = False
            for d in ((step, 0, 0), (-step, 0, 0), (0, step, 0), (0, -step, 0),
                      (0, 0, step), (0, 0, -step)):
                cand = (best[0] + d[0], best[1] + d[1], best[2] + d[2])
                q = _quality(img, cand)
                if q > bq + 1e-4:
                    best, bq, improved = cand, q, True
    return best, bq


def photo_box(img):
    """Returns ((x, y, size), quality) or (None, 0)."""
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    ii = B._integral(gray)
    scored = []
    for cx, cy, cs in _candidates(img):
        x, y, s = B._refine(gray, (cx, cy, cs), rng=max(10, int(0.04 * cs)))
        f = B._checker_score(ii, np.array([[x, y, s]]), max(6, int(s / 60)))[0]
        scored.append((f, (x, y, s)))
    scored.sort(reverse=True)
    best, bq = None, 0.0
    for f, box in scored[:3]:                 # polish the three most board-like
        if f < 0.5:
            continue
        pb, q = _polish(img, box)
        if q > bq:
            best, bq = pb, q
    return best, bq

import cv2, pickle, sys, os, chess, chess.engine
from boardlib import *

# ---------- 1. Read what you typed after "python scripts/game.py" ----------
folder = sys.argv[1]      # folder with the screenshots
tfile = sys.argv[2]       # template file (.pkl), ignored when "ai" is used
turn = sys.argv[3]        # w or b: who moves in the FIRST screenshot

y = None
box = None
start_num = 1                        # move number of the FIRST screenshot
show = "scores" in sys.argv          # show template match scores
use_ai = "ai" in sys.argv            # use the AI model
show_fen = "fen" in sys.argv         # print the board FEN on every line
for a in sys.argv[4:]:
    if a.isdigit():
        y = int(a)                   # a plain number = board Y position
    elif a.startswith("crop="):      # crop=X,Y,SIZE
        box = tuple(int(v) for v in a[5:].split(","))
    elif a.startswith("move="):      # move=3 -> first screenshot is move 3
        start_num = int(a[5:])

DROP_LIMIT = 150                     # 150 = 1.5 pawns; lower = more sensitive
AI_LOW = 0.80                        # AI confidence below this = check by eye


def find_move(prev_board, cur_placement):
    """Which move turns the previous position into the current one?
    Returns it in normal notation (like Nf3), or None if no move fits."""
    pb = prev_board.copy()
    pb.castling_rights = chess.BB_CORNERS        # allow castling to be found
    pb.castling_rights = pb.clean_castling_rights()
    for m in list(pb.legal_moves):
        san = pb.san(m)
        pb.push(m)
        same = pb.board_fen() == cur_placement
        pb.pop()
        if same:
            return san
    return None


def label(num, side, san):
    """5.Qd3 for White, 5...Qxd3 for Black."""
    return f"{num}.{san}" if side == "w" else f"{num}...{san}"


# ---------- 2. Load the recognizer and start Stockfish ----------
if use_ai:
    here = os.path.dirname(os.path.abspath(__file__))
    sys.path.insert(0, os.path.join(here, "..", "ai"))
    from ai_recognize import recognize_ai
    templates = None
else:
    templates = pickle.load(open(tfile, "rb"))

engine = chess.engine.SimpleEngine.popen_uci("stockfish")

files = sorted(f for f in os.listdir(folder)
               if f.lower().endswith((".png", ".jpg", ".jpeg")))
other = "b" if turn == "w" else "w"

prev = None            # evaluation of the previous position
prev_best = None       # best move label in the previous position
prev_board = None      # the previous position as a chess board
prev_side = None       # who was to move in the previous position
prev_num = None        # move number of the previous position
mistakes = []          # list of (move label, pawns lost)

# ---------- 3. Go through the screenshots one by one ----------
for i, name in enumerate(files):
    img = cv2.imread(os.path.join(folder, name))
    if img is None:
        print(name, "can't open")
        prev = prev_best = prev_board = None
        continue

    board = find_board(img, y, box)
    if board is None or board.size == 0:
        print(name, "board not found")
        prev = prev_best = prev_board = None
        continue
    if i == 0:
        cv2.imwrite("output/preview.png", board)

    # screenshot -> grid of pieces
    minc = None
    if use_ai:
        grid, conf = recognize_ai(board)
        minc = min(min(row) for row in conf)      # weakest square
    else:
        grid = recognize_grid(split(board), templates, max_score=40,
                              show=(show and i == 0))

    # grid -> chess position
    if needs_flip(grid):
        grid = flip_grid(grid)
    side = turn if i % 2 == 0 else other          # whose move it is now
    num = start_num + (i + (1 if turn == "b" else 0)) // 2
    b = chess.Board(grid_to_fen(grid, side))
    if not b.is_valid():
        print(name, "invalid position:", b.board_fen())
        prev = prev_best = prev_board = None
        continue

    # ---------- 4. Which move was played to get here? ----------
    played = "start"
    if prev_board is not None:
        san = find_move(prev_board, b.board_fen())
        played = label(prev_num, prev_side, san) if san else "?"

    # ---------- 5. Ask Stockfish ----------
    info = engine.analyse(b, chess.engine.Limit(time=1))
    cur = info["score"].white().score(mate_score=10000)   # + = White better
    best = label(num, side, b.san(info["pv"][0])) if info.get("pv") else None

    line = f"{name}  {played}  eval {cur/100:+.2f}"
    if show_fen:
        line += f"  {b.board_fen()}"
    if minc is not None and minc < AI_LOW:
        line += f"  [ai LOW {minc:.2f}]"

    # ---------- 6. Did that move make things worse? ----------
    if prev is not None:
        mover = "White" if side == "b" else "Black"       # who just moved
        loss = (prev - cur) if side == "b" else (cur - prev)
        if loss > DROP_LIMIT:
            line += (f"   <-- big drop by {mover} ({loss/100:.1f} pawns),"
                     f" better was {prev_best}")
            mistakes.append((played, loss / 100))
    print(line)

    prev, prev_best = cur, best
    prev_board, prev_side, prev_num = b, side, num

engine.quit()

# ---------- 7. Summary ----------
print()
if mistakes:
    print("Mistakes:", ", ".join(f"{m} ({p:.1f})" for m, p in mistakes))
else:
    print("No big drops found.")

import cv2
import numpy as np


# ---------- automatic board finder (checkerboard pattern + edge refine) ----------

def _integral(gray):
    return cv2.integral(gray.astype("float32"))


def _patch_means(ii, xs, ys, size):
    H1, W1 = ii.shape
    x0 = np.clip(xs, 0, W1 - 1 - size).astype(int)
    y0 = np.clip(ys, 0, H1 - 1 - size).astype(int)
    x1, y1 = x0 + size, y0 + size
    return (ii[y1, x1] - ii[y0, x1] - ii[y1, x0] + ii[y0, x0]) / (size * size)


def _checker_score(ii, cand, size):
    """How much does each candidate box look like an 8x8 two-colour checkerboard?"""
    N = len(cand)
    x0 = cand[:, 0][:, None, None].astype(float)
    y0 = cand[:, 1][:, None, None].astype(float)
    sq = cand[:, 2][:, None, None].astype(float) / 8
    r = np.arange(8)[None, :, None]
    c = np.arange(8)[None, None, :]
    xs = np.broadcast_to(x0 + c * sq + 0.12 * sq, (N, 8, 8))
    ys = np.broadcast_to(y0 + r * sq + 0.12 * sq, (N, 8, 8))
    v = _patch_means(ii, xs, ys, size)
    par = ((np.arange(8)[:, None] + np.arange(8)[None, :]) % 2).astype(bool)
    A, B = v[:, ~par], v[:, par]
    mA, mB = np.median(A, axis=1), np.median(B, axis=1)
    contrast = np.abs(mA - mB)
    tol = np.maximum(5.0, 0.15 * contrast)
    good = ((np.abs(A - mA[:, None]) < tol[:, None]).sum(1)
            + (np.abs(B - mB[:, None]) < tol[:, None]).sum(1))
    frac = good / 64.0
    frac[contrast < 12] = 0
    return frac


def _coarse(gray, sc=0.5):
    H, W = gray.shape
    sg = cv2.resize(gray, (int(W * sc), int(H * sc)), interpolation=cv2.INTER_AREA)
    ii = _integral(sg)
    h, w = sg.shape
    cands = []
    for x0 in range(0, int(0.2 * w), 2):
        for S in {w - x0, w - 2 * x0}:      # board touches the right edge, or is centred
            if S < 0.6 * w:
                continue
            for y0 in range(0, h - S + 1, 2):
                cands.append((x0, y0, S))
    if not cands:
        return None, 0.0
    cands = np.array(cands)
    frac = _checker_score(ii, cands, max(4, int(12 * sc / 0.5)))
    j = int(np.argmax(frac))
    top = cands[frac >= frac[j] - 1e-9]
    return np.median(top, axis=0) / sc, float(frac[j])


def _edge_energy(gx, gy, X0, Y0, S, n, W, H):
    ks = np.arange(0, 9)
    sq = S / 8
    t = np.linspace(0, S - 1, n).astype(int)
    hy = np.clip(np.round(Y0 + ks * sq).astype(int), 0, H - 1)
    vx = np.clip(np.round(X0 + ks * sq).astype(int), 0, W - 1)
    return (gy[hy[:, None], (X0 + t)[None, :]].sum()
            + gx[(Y0 + t)[:, None], vx[None, :]].sum())


def _refine(gray, box, rng=18):
    H, W = gray.shape
    x0, y0, S = [int(round(v)) for v in box]
    g = cv2.GaussianBlur(gray, (0, 0), 1.0).astype("float32")
    gx = np.abs(cv2.Sobel(g, cv2.CV_32F, 1, 0, ksize=3))
    gy = np.abs(cv2.Sobel(g, cv2.CV_32F, 0, 1, ksize=3))
    best, bs = (x0, y0, S), -1.0
    for step, span, n in ((2, rng, 48), (1, 3, 64)):   # coarse pass, then fine pass
        cx, cy, cs = best
        for ds in range(-span, span + 1, step):
            for dx in range(-span, span + 1, step):
                for dy in range(-span, span + 1, step):
                    X0, Y0, S2 = cx + dx, cy + dy, cs + ds
                    if X0 < 0 or Y0 < 0 or X0 + S2 > W or Y0 + S2 > H:
                        continue
                    e = _edge_energy(gx, gy, X0, Y0, S2, n, W, H)
                    if e > bs:
                        bs, best = e, (X0, Y0, S2)
    return best


def auto_box(img):
    """Find the board automatically. Returns (x, y, size) or None."""
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    box, frac = _coarse(gray)
    if box is None or frac < 0.75:
        return None
    return _refine(gray, box)


def find_board(img, y=None, box=None):
    H, W = img.shape[:2]
    if box is not None:               # box = (x, y, size)
        bx, by, bs = box
        return img[by:by+bs, bx:bx+bs]
    if y is not None:                 # manual mode: board is full width
        return img[y:y+W, 0:W]
    auto = auto_box(img)              # automatic: look for the checkerboard
    if auto is not None:
        bx, by, bs = auto
        return img[by:by+bs, bx:bx+bs]
    # last resort: the old outline search
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    edges = cv2.dilate(cv2.Canny(gray, 50, 150), None, iterations=2)
    cnts, _ = cv2.findContours(edges, cv2.RETR_LIST, cv2.CHAIN_APPROX_SIMPLE)
    best, area = None, 0
    for c in cnts:
        x, yy, w, h = cv2.boundingRect(c)
        if 0.9 < w / h < 1.1 and w > W * 0.5 and w * h > area:
            best, area = (x, yy, w, h), w * h
    if best is None:
        return None
    x, yy, w, h = best
    s = min(w, h)
    return img[yy:yy+s, x:x+s]


def split(board):
    step = board.shape[0] // 8
    return [[board[r*step:(r+1)*step, c*step:(c+1)*step]
             for c in range(8)] for r in range(8)]


def center(sq):
    h = sq.shape[0]
    m = int(h * 0.15)
    return sq[m:h-m, m:h-m]


def features(sq):
    g = cv2.cvtColor(center(sq), cv2.COLOR_BGR2GRAY)
    g = cv2.resize(g, (32, 32))
    e = cv2.Canny(g, 50, 150)
    return cv2.GaussianBlur(e, (5, 5), 0).astype("float32")


def is_empty(sq):
    g = cv2.cvtColor(center(sq), cv2.COLOR_BGR2GRAY)
    return g.std() < 12


def recognize_grid(squares, templates, max_score=15, show=False):
    grid = []
    for r in range(8):
        row = []
        for c in range(8):
            sq = squares[r][c]
            if is_empty(sq):
                row.append(".")
                continue
            f = features(sq)
            best, best_score = "?", 1e9
            for piece, plist in templates.items():
                for t in plist:
                    score = abs(f - t).mean()
                    if score < best_score:
                        best, best_score = piece, score
            if show:
                print(f"{r}{c} {best} {best_score:.1f}")
            if max_score is not None and best_score > max_score:
                print(f"  dropped {r}{c} {best} {best_score:.1f}")
                row.append(".")
            else:
                row.append(best)
        grid.append(row)
    return grid


def grid_to_fen(grid, turn="w"):
    rows = []
    for row in grid:
        s, empty = "", 0
        for p in row:
            if p == ".":
                empty += 1
            else:
                if empty:
                    s += str(empty)
                    empty = 0
                s += p
        if empty:
            s += str(empty)
        rows.append(s)
    return "/".join(rows) + f" {turn} - - 0 1"


def flip_grid(grid):
    return [row[::-1] for row in grid[::-1]]


def needs_flip(grid):
    w_rows, b_rows = [], []
    for r in range(8):
        for c in range(8):
            p = grid[r][c]
            if p == ".":
                continue
            (w_rows if p.isupper() else b_rows).append(r)
    if not w_rows or not b_rows:
        return False
    return sum(w_rows) / len(w_rows) < sum(b_rows) / len(b_rows)

"""One entry point for reading a chess screenshot.
The command-line scripts and (later) the Android app both call this file.
It never prints; it returns a dictionary."""
import os, sys, pickle, cv2

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, HERE)
from boardlib import (find_board, split, recognize_grid, grid_to_fen,
                      flip_grid, needs_flip)

LOW_CONF = 0.80          # AI confidence below this = check the board by eye


def _photo_fallback(img):
    """Slower search for a small board inside a bigger picture (AI mode only)."""
    sys.path.insert(0, os.path.join(HERE, "..", "ai"))
    from photo_board import photo_box
    box, quality = photo_box(img)
    if box is None or quality < 0.6:
        return None
    x, y, s = box
    return img[y:y+s, x:x+s]


def read_board(img, use_ai=True, templates=None, crop=None, y=None, flip=None):
    """Screenshot (OpenCV image) -> grid of pieces.
    flip: None = auto, True = force flip, False = never flip."""
    board = find_board(img, y, crop)
    if (board is None or board.size == 0) and use_ai and crop is None and y is None:
        board = _photo_fallback(img)         # e.g. a photo of a screen
    if board is None or board.size == 0:
        return {"ok": False, "error": "Board not found"}

    conf = None
    if use_ai:
        sys.path.insert(0, os.path.join(HERE, "..", "ai"))
        from ai_recognize import recognize_ai
        grid, conf = recognize_ai(board)
    else:
        if templates is None:
            return {"ok": False, "error": "No templates given"}
        grid = recognize_grid(split(board), templates, max_score=40)

    do_flip = needs_flip(grid) if flip is None else flip
    if do_flip:
        grid = flip_grid(grid)
        if conf:
            conf = flip_grid(conf)
    low = min(min(r) for r in conf) if conf else None
    return {"ok": True, "grid": grid, "conf": conf, "min_conf": low,
            "board": board}


def analyze_image(path, turn="w", use_ai=True, templates_file=None,
                  crop=None, y=None, flip=None, think=3):
    """Screenshot file -> FEN, best move and score. Returns a dictionary:
    ok, error, fen, valid, min_conf, low_conf, best_move, score"""
    img = cv2.imread(path)
    if img is None:
        return {"ok": False, "error": "Can't open " + str(path)}

    templates = None
    if not use_ai:
        try:
            templates = pickle.load(open(templates_file, "rb"))
        except Exception as e:
            return {"ok": False, "error": "Templates: %s" % e}

    r = read_board(img, use_ai, templates, crop, y, flip)
    if not r["ok"]:
        return r

    import chess, chess.engine           # only needed from here on
    fen = grid_to_fen(r["grid"], turn)
    b = chess.Board(fen)
    out = {"ok": True, "error": None, "fen": fen, "board_text": str(b),
           "valid": b.is_valid(), "min_conf": r["min_conf"],
           "low_conf": r["min_conf"] is not None and r["min_conf"] < LOW_CONF,
           "best_move": None, "score": None}
    if not out["valid"]:
        out["error"] = "Invalid position: check the recognition first"
        return out

    path_sf = os.environ.get("STOCKFISH_PATH", "stockfish")
    try:
        engine = chess.engine.SimpleEngine.popen_uci(path_sf)
    except Exception as e:
        out["error"] = "Stockfish not found (%s)" % e
        return out
    try:
        info = engine.analyse(b, chess.engine.Limit(time=think))
    finally:
        engine.quit()
    out["best_move"] = b.san(info["pv"][0]) if info.get("pv") else None
    out["score"] = str(info["score"].pov(b.turn))
    return out

def board_svg(fen, best, turn):
    """Draw the position the program read, with an arrow for the best move."""
    try:
        import chess, chess.svg
        b = chess.Board(fen)
        arrows = []
        if best:
            m = b.parse_san(best)
            arrows = [chess.svg.Arrow(m.from_square, m.to_square, color="#2e8b57cc")]
        return chess.svg.board(b, size=340, arrows=arrows, flipped=(turn == "b"))
    except Exception:
        return None

    """Save upload to a temp file, run pipeline, and clean up."""
    fd, path = tempfile.mkstemp(suffix=".img")
    os.close(fd)
    try:
        file_storage.save(path)
        flip = {"flip": True, "noflip": False}.get(flip_raw)
        res = pipeline.analyze_image(path, turn, use_ai, TEMPLATES, None, None, flip)
    finally:
        if os.path.exists(path):
            os.remove(path)
    return res

import os
import cv2
import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
MODEL = os.path.join(HERE, "models", "chess-tiles-v2.onnx")
LABELS = "1KQRBNPkqrbnp"          # index 0 = empty square

_net = None


def _load():
    global _net
    if _net is None:
        if not os.path.exists(MODEL):
            raise FileNotFoundError("Model not found: " + MODEL)
        _net = cv2.dnn.readNetFromONNX(MODEL)
    return _net


def recognize_ai(board):
    """board: BGR image that contains exactly the 8x8 board.
    Returns (grid, conf): 8x8 lists of piece letters ('.' = empty)
    and of confidences (0 to 1), top-left square first."""
    gray = cv2.cvtColor(board, cv2.COLOR_BGR2GRAY)
    gray = cv2.resize(gray, (256, 256), interpolation=cv2.INTER_LINEAR)
    gray = gray.astype("float32") / 255.0

    tiles = np.zeros((64, 1024), dtype="float32")
    for r in range(8):
        for c in range(8):
            t = gray[r*32:(r+1)*32, c*32:(c+1)*32]
            tiles[r*8 + c] = t.reshape(1024)

    net = _load()
    net.setInput(tiles)
    out = np.array(net.forward()).reshape(64, 13)

    # make sure rows are probabilities (they should already be)
    if out.min() < 0 or abs(out.sum(axis=1).mean() - 1.0) > 0.05:
        e = np.exp(out - out.max(axis=1, keepdims=True))
        out = e / e.sum(axis=1, keepdims=True)

    grid, conf = [], []
    for r in range(8):
        grow, crow = [], []
        for c in range(8):
            p = out[r*8 + c]
            k = int(p.argmax())
            grow.append("." if k == 0 else LABELS[k])
            crow.append(float(p[k]))
        grid.append(grow)
        conf.append(crow)
    return grid, conf


def weak_squares(conf, limit=0.9):
    """List of (row, col, confidence) below the limit."""
    return [(r, c, conf[r][c]) for r in range(8) for c in range(8)
            if conf[r][c] < limit]
